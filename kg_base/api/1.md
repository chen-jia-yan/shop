:param id => '把 node_id 粘贴在这里'


// 1. 这个节点在不在、在哪个 scope、是什么类型
MATCH (n {id: $id}) RETURN labels(n), n.scope, n.label, n.type;


// 2. 它和切片之间有哪些关系：关系名、方向、切片的 scope
MATCH (n {id: $id})-[r]-(c:Chunk)
RETURN type(r) AS 关系名,
       CASE WHEN startNode(r) = n THEN '实体→切片' ELSE '切片→实体' END AS 方向,
       c.scope AS 切片scope, count(*) AS 数量;


// 3. 整个 jp-poc 里，实体和切片之间的关系是什么样的
MATCH (e:Entity {scope: 'jp-poc'})-[r]-(c:Chunk)
RETURN type(r) AS 关系名,
       CASE WHEN startNode(r) = e THEN '实体→切片' ELSE '切片→实体' END AS 方向,
       c.scope AS 切片scope, count(*) AS 数量;


// 4. jp-poc 的切片总数，以及有多少带了摘要
MATCH (c:Chunk {scope: 'jp-poc'})
RETURN count(c) AS 切片数, count(c.preview) AS 有摘要的切片数;







MATCH (e:Entity {scope: 'jp-poc'})-[:MENTIONED_IN]->(c:Chunk)
RETURN e.id AS id, e.label AS 名称, e.type AS 类型, count(c) AS 切片数
ORDER BY 切片数 DESC LIMIT 5;
