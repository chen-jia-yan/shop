```python
@dataclass(frozen=True)
class SourceDocument:
    """A single input document source."""

    filename: str
    path: Path | None = None
    stream: BinaryIO | None = None
    content: bytes | None = None
    content_type: str | None = None
    metadata: Mapping[str, object] = field(default_factory=dict)
    idempotency_key: str | None = None

    def __post_init__(self) -> None:
        if self.path is None and self.stream is None and self.content is None:
            raise ValueError("One of path, stream, or content must be provided")

    @classmethod
    def from_path(cls, path: Path | str, **kwargs: Any) -> SourceDocument:
        path = Path(path)
        return cls(
            filename=kwargs.pop("filename", path.name),
            path=path,
            **kwargs,
        )

    @classmethod
    def from_stream(
        cls,
        stream: BinaryIO,
        filename: str,
        **kwargs: Any,
    ) -> SourceDocument:
        return cls(filename=filename, stream=stream,** kwargs)

    @classmethod
    def from_bytes(
        cls,
        data: bytes,
        filename: str,
        **kwargs: Any,
    ) -> SourceDocument:
        return cls(filename=filename, content=data, **kwargs)
```

```python
def compute_document_id(path: Path) -> str:
    """Compute a stable document_id from file content hash."""
    digest = hashlib.sha256()
    with path.open("rb") as f:
        while True:
            block = f.read(1024 * 1024)
            if not block:
                break
            digest.update(block)
    return f"doc::{digest.hexdigest()[:16]}"


def compute_version_id(document_id: str, pipeline_config_hash: str) -> str:
    """Compute a stable version_id from document_id and pipeline config hash."""
    raw = f"{document_id}:{pipeline_config_hash}"
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()[:16]
```
