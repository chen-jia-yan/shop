```python
def _chunk_relations(self, chunk: DocumentChunk, lexicon: Lexicon) -> Iterator[RelationSpec]:
    cid = chunk.metadata.chunk_id
    flow_nodes: list[EntityRef] = []
    flowchart = chunk.metadata.flowchart_data
    if isinstance(flowchart, dict):
        flow_nodes = yield from _flowchart_relations(flowchart, cid, lexicon)

    stages = lexicon.concepts_in(chunk.content, STAGE)
    if len(stages) == 1:
        stage = concept_ref(stages[0].concept)
        actions = [concept_ref(m.concept) for m in lexicon.concepts_in(chunk.content, ACTION)]
        for node in dict.fromkeys([*flow_nodes, *actions]):
            yield RelationSpec(node, "PART_OF", stage, cid, "rule")

    if self._cooccurrence_roles:
        yield from _cooccurrence_relations(chunk.content, cid, lexicon)


def _flowchart_relations(flowchart: dict[str, Any], chunk_id: str, lexicon: Lexicon) -> Iterator[RelationSpec]:
    """产出流程图关系；生成器返回值为该图中的流程节点。"""
    by_id: dict[Any, EntityRef] = {}
    for node in flowchart.get("nodes") or []:
        label = clean_label(str(node.get("text") or ""))
        if not label or len(label) > _MAX_NODE_LABEL:
            continue
        ref = flow_node_ref(label, lexicon)
        by_id[node.get("node_id")] = ref
        lane = clean_label(str(node.get("lane") or ""))
        if lane and len(lane) <= _MAX_LANE_LABEL:
            yield RelationSpec(ref, "HANDLED_BY", lane_ref(lane, lexicon), chunk_id, "flowchart")
    for edge in flowchart.get("edges") or []:
        src = by_id.get(edge.get("source_id"))
        dst = by_id.get(edge.get("target_id"))
        if src is not None and dst is not None and src != dst:
            yield RelationSpec(src, "NEXT", dst, chunk_id, "flowchart")
    return list(dict.fromkeys(by_id.values()))


def _cooccurrence_relations(text: str, chunk_id: str, lexicon: Lexicon) -> Iterator[RelationSpec]:
    for sentence in _SENTENCE_SPLIT.split(text):
        # find 已去掉被更长词包含的短词（「审核员」里的「审核」不会被当成动作）。
        roles = lexicon.concepts_in(sentence, ROLE)
        if not roles:
            continue
        for action in lexicon.concepts_in(sentence, ACTION):
            for role in roles:
                yield RelationSpec(
                    concept_ref(action.concept), "HANDLED_BY", concept_ref(role.concept), chunk_id, "rule"
                )
```

```python
def build_messages(document: ProcessedDocument, max_chars: int = MAX_PROMPT_CHARS) -> list[dict[str, str]]:
    blocks: list[str] = []
    total = 0
    for i, chunk in enumerate(document.chunks, start=1):
        block = f"[Chunk {i}] {document.filename}\n{chunk.content}"
        remaining = max_chars - total
        if len(block) > remaining:
            if remaining <= 200:
                break
            block = block[:remaining] + "\n...(截断)"
        blocks.append(block)
        total += len(block)
    user = "请从以下文档 Chunk 中抽取业务实体和关系：\n\n" + "\n\n---\n\n".join(blocks)
    return [{"role": "system", "content": _SYSTEM_PROMPT}, {"role": "user", "content": user}]
```

```python
def upsert(self, document: ProcessedDocument) -> SinkResult:
    url = self._resolve_url(document)
    if url is None:
        logger.warning("文档 %s 没有 url，Copilot Studio 引用将不可用 (设置 source_url_base 或 url_resolver)", document.filename)

    indexed = self._repo.finalize_document_upsert(document.document_id, document.version_id)
```
