## 5. 部署步骤

### 5.1 第一阶段：基础设施

```bash
# 1. 创建资源组
az group create --name rg-aittest-shared --location eastus

# 2. 创建存储账户
az storage account create \
  --name staittestshared001 \
  --resource-group rg-aittest-shared \
  --sku Standard_LRS \
  --kind StorageV2

# 3. 创建 Azure Container Registry
az acr create \
  --name acraittest001 \
  --resource-group rg-aittest-shared \
  --sku Basic \
  --admin-enabled true

# 4. 创建 Key Vault
az keyvault create \
  --name kv-aittest-shared \
  --resource-group rg-aittest-shared \
  --sku Standard

# 5. 创建 Log Analytics Workspace
az monitor log-analytics workspace create \
  --resource-group rg-aittest-shared \
  --workspace-name law-aittest-shared

# 6. 创建 Application Insights
az monitor app-insights component create \
  --app ai-aittest-shared \
  --location eastus \
  --resource-group rg-aittest-shared \
  --workspace law-aittest-shared
```

### 5.2 第二阶段：环境部署（以 Dev 为例）

```bash
# 1. 创建资源组
az group create --name rg-aittest-dev --location eastus

# 2. 创建 Azure Functions（带 Application Insights）
az functionapp create \
  --name func-aittest-dev-001 \
  --resource-group rg-aittest-dev \
  --consumption-plan-location eastus \
  --runtime python \
  --runtime-version 3.11 \
  --functions-version 4 \
  --storage-account staittestdev001 \
  --app-insights ai-aittest-shared \
  --os-type Linux

# 3. 配置 Functions 环境变量
az functionapp config appsettings set \
  --name func-aittest-dev-001 \
  --resource-group rg-aittest-dev \
  --settings \
    AZURE_OPENAI_ENDPOINT="$(az keyvault secret show --name AzureOpenAIEndpoint --vault-name kv-aittest-shared --query value -o tsv)" \
    APPINSIGHTS_INSTRUMENTATIONKEY="$(az monitor app-insights component show --app ai-aittest-shared --resource-group rg-aittest-shared --query instrumentationKey -o tsv)"

# 4. 创建 SQL Database
az sql server create \
  --name sqlsvr-aittest-dev \
  --resource-group rg-aittest-dev \
  --admin-user aittestadmin \
  --admin-password "$(az keyvault secret show --name SQLAdminPassword --vault-name kv-aittest-shared --query value -o tsv)"

az sql db create \
  --name sqldb-aittest-dev \
  --server sqlsvr-aittest-dev \
  --resource-group rg-aittest-dev \
  --service-objective S0

# 5. 创建 Azure AI Search
az search service create \
  --name srch-aittest-dev \
  --resource-group rg-aittest-dev \
  --sku Standard \
  --partition-count 1 \
  --replica-count 1

# 6. 创建 Service Bus
az servicebus namespace create \
  --name sbns-aittest-dev \
  --resource-group rg-aittest-dev \
  --sku Standard

az servicebus queue create \
  --namespace-name sbns-aittest-dev \
  --name ait-test-queue \
  --resource-group rg-aittest-dev
```

### 5.3 第三阶段：Power Platform 部署

```bash
# 前置条件：安装 Power Platform CLI (pac)
# https://docs.microsoft.com/en-us/power-platform/developer/cli/introduction

# 1. 认证

# 2. 从源环境导出 Solution

  --path ./deployment/solutions \
  --name ait_TestPlatform \
  --managed false \
  --include-activelayer-customizations true

# 3. 拆包为源码
pac solution unpack \
  --zipfile ./deployment/solutions/ait_TestPlatform.zip \
  --folder ./deployment/solutions/src

# 4. 导入到目标环境
pac solution import \
  --path ./deployment/solutions/ait_TestPlatform.zip \
  --publish-workflows true \
  --activate-plugins true

pac solution env-variable set \
  --variable-name ait_AzureOpenAIEndpoint \

# 6. 发布所有自定义
pac solution publish
```

## 6. 回滚策略

### 6.1 回滚类型

| 类型 | 触发条件 | 策略 | 预估耗时 |
|---|---|---|---|
| Functions 代码回滚 | CI/CD 部署后功能异常 | 重新部署上一个版本 Image | 5-10 分钟 |
| Database 回滚 | 数据迁移失败 | 执行 `dotnet ef migrations remove` + 还原备份 | 15-30 分钟 |
| Power Platform 回滚 | Solution 导入异常 | 导入上一个 Managed Solution 版本 | 10-20 分钟 |
| AI 模型回滚 | 模型版本质量下降 | 切换部署到上一版本 | 5 分钟 |
| 完整回滚 | 灾难性故障 | 从备份还原整环境 | 2-4 小时 |

### 6.2 代码回滚

```bash
# Azure Functions - 回滚到上一个版本
# 方式 1：通过 Deployment Slots（推荐）
az webapp deployment slot swap \
  --name func-aittest-prod-001 \
  --resource-group rg-aittest-prod \
  --slot staging \
  --action swap

# 方式 2：通过 ACR tag 回滚
az functionapp config container set \
  --name func-aittest-prod-001 \
  --resource-group rg-aittest-prod \
  --docker-custom-image-name acraittest001.azurecr.io/ai-test-functions:v1.0.3-previous

# 方式 3：通过 Azure DevOps Pipeline 重新运行上一个成功版本
# 在 Pipelines → Runs → 选择上一个成功版本 → Re-run
```

### 6.3 数据库回滚

```bash
# 方式 1：时间点还原（Point-in-Time Restore）
az sql db restore \
  --resource-group rg-aittest-prod \
  --server sqlsvr-aittest-prod \
  --name sqldb-aittest-prod \
  --dest-name sqldb-aittest-prod-restored \
  --time "2026-08-17T14:00:00"

# 方式 2：EF Core 迁移回滚
dotnet ef database update PreviousMigrationName \
  --connection "$CONNECTION_STRING"

# 方式 3：从自动备份创建新库并切换连接字符串
az sql db restore \
  --dest-name sqldb-aittest-prod-restored \
  --edition Standard \
  --service-objective S3 \
  --resource-group rg-aittest-prod \
  --server sqlsvr-aittest-prod \
  --name sqldb-aittest-prod \
  --deleted-time "$(az sql db list-deleted --server sqlsvr-aittest-prod --resource-group rg-aittest-prod --query "[0].deletionTime" -o tsv)"
```

#### 关键监控指标

| OpenAI API 调用错误 | > 1% | Critical |

#### 自定义事件埋点（代码级）

### 7.2 Log Analytics 工作区配置

```bash
# 创建查询 - 错误趋势
# 保存为：Error Trend Query
cat << 'EOF' > ./monitoring/queries/error-trend.kql
union requests, exceptions
| where timestamp > ago(24h)
| summarize Count = count() by bin(timestamp, 1h), type = iif(itemType == "request", "Request", "Exception")
| render timechart
EOF

# 创建查询 - Functions 性能
cat << 'EOF' > ./monitoring/queries/function-performance.kql
requests
| where cloud_RoleName == "ai-test-functions"
| where timestamp > ago(1h)
| summarize
    AvgDuration = avg(duration),
    P95Duration = percentile(duration, 95),
    P99Duration = percentile(duration, 99),
    FailureRate = countif(success == false) * 100.0 / count()
  by name
| sort by AvgDuration desc
EOF

# 创建查询 - OpenAI 调用统计
cat << 'EOF' > ./monitoring/queries/openai-usage.kql
dependencies
| where type == "HTTP"
| where target contains "openai.azure.com"
| where timestamp > ago(24h)
| summarize
    TotalCalls = count(),
    AvgLatency = avg(duration),
    TotalTokens = sum(customMeasurements['prompt_tokens'] + customMeasurements['completion_tokens'])
  by bin(timestamp, 1h)
| render timechart
EOF
```
