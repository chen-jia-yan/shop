## 1  接口约定
class EvidenceItem(BaseModel):
    kind: Literal["chunk", "kg_fact", "derived"]  # 原文切片 / 图谱事实 / 推断(复原、流程重建)
    content: str
    chunk_id: str | None = None
    source_file: str | None = None
    page_or_sheet: str | None = None
    score: float | None = None

class QaResult(BaseModel):
    answer: str
    evidence: list[EvidenceItem]   # 答案实际用到的全部证据,判官只看这一份
    trace_id: str | None = None

class QaAnswerPort(Protocol):
    async def answer(self, *, question: str, mode: Literal["llm", "agent"], top_k: int) -> QaResult: ...

class ChunkSourcePort(Protocol):     # 切片审阅读数据(只读)
    async def list_kbs(self) -> list[KbSummary]: ...
    async def list_files(self, *, kb_id: str | None, keyword: str | None, page: int, page_size: int) -> tuple[list[FileSummary], int]: ...
    async def get_file(self, *, file_id: str) -> FileDetail | None: ...
    async def list_chunks(self, *, file_id: str, page: int, page_size: int) -> tuple[list[ChunkView], int]: ...
    async def get_artifact(self, *, artifact_id: str) -> ArtifactView | None: ...
    async def preview_chunks(self, *, file_id: str, strategy: dict[str, Any]) -> list[ChunkView]: ...

class ChunkPublishPort(Protocol):    # 审阅后落库(写入只在同事那边发生)
    async def publish(self, *, file_id: str, chunks: list[ChunkEdit], operator: str | None) -> PublishResult: ...

## p3 评估服务与接口

   - start_run(req)：校验评估集存在、问答服务已配置、判官模型已配置；写入 run（queued）和每道题的 item（pending）；
     在后台启动执行任务，并保存任务引用（防止被垃圾回收）；返回 run_id
   - 后台执行：
     · run 改为 running，记录 started_time
     · 用信号量控制同时调模型的数量（LIBRARY_EVAL_MAX_CONCURRENCY）
     · 每道题：QaAnswerPort.answer() → 用其返回的 evidence 构造判官消息 → ModelGatewayPort.acompletion(temperature=0)，
       把流式结果拼成完整文本 → parse_judge_output → 开关打开时做来源可追溯检查 → analyzer 分类 → 写 item → done+1、更新 heartbeat_time

     · 单题出错：item 标为 error，failed_count+1，继续下一题
     · 全部结束：计算 summary，run 标为 succeeded；遇到致命错误标 failed 并记录 error
   - cancel_run：只设置 cancel_requested=true
   - 启动恢复：应用启动时，把 status 为 running 且 heartbeat_time 超过 LIBRARY_EVAL_HEARTBEAT_TIMEOUT_SEC 的 run 标为 interrupted
     （心跳还新鲜的不要动，它可能正在另一个实例上运行）
   - 查询：run 详情、run 列表、item 列表（可按 category 筛选）、item 详情、分析结果（由 item 现算）
