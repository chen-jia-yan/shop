```python
async def upload(
    self,
    *,
    conv_id: str | None,
    filename: str | None,
    content_type: str | None,
    read: Callable[[int], Awaitable[bytes]],
) -> dict:
    """上传附件: 可选校验会话，流式写存储，落元数据，返回 file_id。

    `conv_id` 给定时校验会话存在并直接归属；否则暂存（延迟绑定）。
    """
    storage = self._storage
    if storage is None:
        raise BusinessError(code="CCB001", message="存储未配置")
    if conv_id:
        async with self._session_factory() as session:
            conv = await ConversationRepo(session).get(conv_id)
            if conv is None:
                raise BusinessError(code="CCB001", message="会话不存在")

    file_id = uuid.uuid4().hex
    # 存储路径用 file_id 命名的安全附件名（只含 file_id + 白名单扩展名），
    # 用户原始 filename 不进入路径；filename 原样存元数据（见 add）。
    path = f"ai/{safe_attachment_name(file_id, content_type, filename)}"

    size_bytes = 0

    async def _chunks():
        nonlocal size_bytes
        while True:
            chunk = await read(STORAGE_CHUNK_SIZE_BYTES)
            if not chunk:
                break
            size_bytes += len(chunk)
            yield chunk

    await storage.upload_stream(
        path=path,
        stream=_chunks(),
        content_type=content_type or None,
    )

    async with self._session_factory() as session:
        await AttachmentRepo(session).add(
            id=file_id,
            conversation_id=conv_id or None,
            filename=filename or "unnamed",
            path=path,
            content_type=content_type,
            size_bytes=size_bytes,
        )
        # 审计上传动作（附件元数据先落；审计自成一事务，上传即留痕）。
        await self._audit_factory(session).record_trace(
            _upload_audit_event(
                file_id=file_id,
                conv_id=conv_id,
            ),
        )

    return {
        "file_id": file_id,
        "filename": filename or "unnamed",
        "size": size_bytes,
        "mime": content_type,
    }
```

## 5. core/lifecycle.py 的完整原文：如何登记资源、关闭时怎么调用
@dataclass
class Lifecycle:
    """应用级资源清理注册表（逆序关闭 + 超时 + 异常收集）。"""

    close_timeout_seconds: float = 5.0
    _callbacks: list[tuple[CloseCallback, float | None]] = field(default_factory=list)
    _closed: bool = False

    # ------------------------------------------------
    # 注册
    # ------------------------------------------------
    def push(self, resource: object) -> object:
        """登记一个资源对象；若带有可关闭方法（aclose/close/dispose）则自动包装。

        返回原资源，便于 ``self._redis = lifecycle.push(redis)`` 链式写法。
        关闭方法优先级: ``aclose`` → ``close`` → ``dispose``。
        资源既无 ``aclose`` 也无 ``close``/``dispose`` 时抛出错误（fail 明确），
        避免静默登记了一个"根本不会被关闭"的东西。
        """
        for name in ("aclose", "close", "dispose"):
            closer = getattr(resource, name, None)
            if callable(closer):
                return self.callback(closer)
        raise TypeError(
            f"resource {type(resource).__name__} has no async close operation"
        )

    def callback(
        self,
        callback: CloseCallback,
        *,
        timeout: float | None = None,
    ) -> Callable[[], Awaitable[object] | object]:
        """登记一个清理回调；返回原回调（便于链式）。

        ``timeout`` 覆盖该回调的等待上限。不传则用 ``close_timeout_seconds``。
        """
        if self._closed:
            raise RuntimeError("lifecycle is already closed")
        if not callable(callback):
            raise TypeError("cleanup callback must be callable")
        self._callbacks.append((callback, timeout))
        return callback

    # ------------------------------------------------
    # 关闭
    # ------------------------------------------------
    async def close(self) -> None:
        """逆序执行所有清理回调，每个最多等待 ``close_timeout_seconds``。

        - 逆序: 后创建的资源先关闭（通常是依赖关系的反序）。
        - 超时: 单个清理卡死会被终止并记录，推进到其余清理。
        - 异常: 收集为 :class:`ExceptionGroup` / :class:`BaseExceptionGroup` 抛出。
        - 幂等: 重复调用直接返回。
        """
        if self._closed:
            return
        self._closed = True
        callbacks = list(reversed(self._callbacks))
        self._callbacks.clear()

        errors: list[BaseException] = []
        for callback, timeout in callbacks:
            limit = self.close_timeout_seconds if timeout is None else timeout
            try:
                result = callback()
                if inspect.isawaitable(result):
                    await asyncio.wait_for(result, timeout=limit)
            except TimeoutError:
                errors.append(_LifecycleTimeout(callback))
            except (
                RuntimeError,
                OSError,
                TypeError,
                ValueError,
                AttributeError,
                AssertionError,
                asyncio.CancelledError,
            ) as error:
                errors.append(error)

        if len(errors) == 1:
            raise errors[0]
        if errors:
            if all(isinstance(error, Exception) for error in errors):
                excs = cast(list[Exception], errors)
                raise ExceptionGroup("tap_ai lifecycle close failed", excs)
            raise BaseExceptionGroup("tap_ai lifecycle close failed", errors)


class _LifecycleTimeout(BaseException):
    """单个清理回调超时（不属于业务异常，单独标记）。"""

    def __init__(self, callback: CloseCallback) -> None:
        self.callback = callback
        super().__init__("lifecycle close callback timed out")


# 应用级单例。bootstrap 在应用启动时把引擎 / Redis / 网关等资源登记进来，
# 关闭时调用 :meth:`close`。放在这里（而非容器内）避免循环导入。
lifecycle: Lifecycle = Lifecycle()

@asynccontextmanager
async def _lifespan(_app: object) -> AsyncIterator[None]:
    """应用启动清理孤儿附件；关闭时统一关闭登记过的运行时资源（连接池 / Redis 等）。

    - 启动: 清理"上传后从未绑定会话、已超时"的暂存附件（元数据 + Blob 字节），
      避免无会话上传积累杂质文件。
    - 关闭: 逆序关闭 `lifecycle` 里登记的资源。
    """
    from tap_ai.utils.workspace import reset_workspace_root

    app_state = cast(Any, _app).state if _app is not None else None
    if app_state is not None:
        file_service = getattr(app_state, "file_service", None)
        if file_service is not None:
            try:
                await file_service.cleanup_orphan_files()
            except Exception: # 启动清理失败不阻断应用启动
                _tap_logger.warning("启动时清理孤儿附件失败", exc_info=True)

    await reset_workspace_root()

    # 预置 skill 同步（幂等、失败不阻断）。
    if app_state is not None:
        skill_sync_service = getattr(app_state, "skill_sync_service", None)
        if skill_sync_service is not None:
            try:
                await skill_sync_service.sync()
            except Exception: # strict 模式下失败会抛出，此处兜底记录
                _tap_logger.warning("预置 skill 同步失败", exc_info=True)
    # 同步完成后刷新 Skill 索引: 装配期目录可能尚不存在或为空，
    # 必须在同步后重建索引，否则 L1 目录对模型始终为空。
    skill_catalog = getattr(app_state, "skill_catalog", None)
    if skill_catalog is not None:
        try:
            indexed = skill_catalog.refresh()
            _tap_logger.info("Skill 索引刷新完成: %d 个", len(indexed))
        except Exception:
            _tap_logger.warning("Skill 索引刷新失败", exc_info=True)

    # 模型能力联调验证（失败不阻断启动；未验证能力在运行期按保守值处理）。
    if app_state is not None:
        capability_verify = getattr(app_state, "capability_verify", None)
        if capability_verify is not None:
            try:
                await capability_verify()
            except Exception:
                _tap_logger.warning("模型能力验证失败", exc_info=True)

    # 预热模型网关连接池，避免首个用户请求承担 TCP/TLS 建连开销。
    if app_state is not None:
        container = getattr(app_state, "container", None)
        registry = getattr(container, "model_registry", None)
        gateway = registry.get("model_gateway") if registry is not None else None
        warmup = getattr(gateway, "warmup", None)
        if warmup is not None:
            try:
                await warmup()
                _tap_logger.info("模型网关连接预热完成")
            except Exception:
                _tap_logger.warning("模型网关连接预热失败", exc_info=True)

    if app_state is not None:
        library_eval_service = getattr(app_state, "library_eval_service", None)
        if library_eval_service is not None:
            try:
                await library_eval_service.recover_stale_runs()
            except Exception:
                _tap_logger.warning("Library 评估中断恢复失败", exc_info=True)

    try:
        yield
    finally:
        await lifecycle.close()
