D. 插座标准（contracts/library.py，原样采用；字段说明写成中文 docstring）
from typing import Any, Literal, Protocol
from pydantic import BaseModel, Field

AnswerMode = Literal["llm", "agent"]                    # llm：检索+一次生成；agent：智能体
EvidenceKind = Literal["chunk", "kg_fact", "derived"]   # 原文切片 / 图谱事实 / 推断内容

class EvidenceItem(BaseModel):
    kind: EvidenceKind
    content: str
    chunk_id: str | None = None
    source_file: str | None = None
    page_or_sheet: str | None = None
    score: float | None = None

class QaResult(BaseModel):
    answer: str
    evidence: list[EvidenceItem] = Field(default_factory=list)  # 生成答案时实际用到的全部证据
    trace_id: str | None = None

class QaAnswerPort(Protocol):
    async def answer(self, *, question: str, mode: AnswerMode, top_k: int) -> QaResult: ...

class FileSummary(BaseModel):
    file_id: str            # 文档 id，格式如 doc::1a2b3c4d5e6f7a8b
    source_file: str
    doc_type: str | None = None
    chunk_count: int

class FileDetail(FileSummary):
    locators: list[str] = Field(default_factory=list)   # 该文件的 sheet 名或 slide 列表

class ChunkView(BaseModel):
    chunk_id: str
    file_id: str
    source_file: str
    page_or_sheet: str | None = None
    part_index: int | None = None
    content: str
    metadata: dict[str, Any] = Field(default_factory=dict)

class ChunkSourcePort(Protocol):
    async def list_files(self, *, keyword: str | None, page: int, page_size: int) -> tuple[list[FileSummary], int]: ...
    async def get_file(self, *, file_id: str) -> FileDetail | None: ...
    async def list_chunks(self, *, file_id: str, page: int, page_size: int) -> tuple[list[ChunkView], int]: ...

DraftChunkOrigin = Literal["source", "edited", "added"]

class DraftChunk(BaseModel):
    chunk_id: str | None = None      # 新增的切片为 None
    page_or_sheet: str | None = None
    content: str
    origin: DraftChunkOrigin

class PublishResult(BaseModel):
    success: bool
    published_chunk_count: int = 0
    message: str | None = None

class ChunkPublishPort(Protocol):
    async def publish(self, *, file_id: str, chunks: list[DraftChunk], operator: str | None) -> PublishResult: ...
    # 语义：用 chunks 整体替换该文件在索引中的全部切片

I. 状态
- 草稿：draft（草稿）→ submitted（落库中）→ published（已落库）/ publish_failed（落库失败）
  · 可修改：draft、publish_failed（修改后回到 draft）
  · 可提交：draft、publish_failed
  · 可删除：除 submitted 以外的状态（软删除）
- 评估任务：queued → running → succeeded / failed / cancelled / interrupted
- 评估题目：pending → done / error
- done = 已结束的题目数（成功和出错都算）；failed_count = 出错的题目数
